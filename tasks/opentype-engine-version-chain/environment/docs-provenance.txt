How environment/docs/ was produced: the build stage below fetches each
document from its official source, checks it against the pinned sha256,
and makes the text conversions. The resulting files are stored in docs/
and copied into the image directly. Re-run these steps to refresh a
document.

# Documentation for the offline agent, fetched from the official sources
# listed in docs-manifest.txt, each pinned by sha256 (for kind "md", of the
# page with its site-metadata front matter removed). The final image receives
# /opt/docs and nothing else from this stage.
FROM node:22-bookworm-slim AS docs
RUN apt-get update \
    && apt-get install -y --no-install-recommends ca-certificates curl \
    && rm -rf /var/lib/apt/lists/*
COPY docs-manifest.txt /opt/docs/SOURCES.txt
RUN set -eu; \
    cd /opt/docs; \
    re='^[0-9a-f]\{64\}  '; \
    : > /tmp/docs.sha256; \
    grep "$re" SOURCES.txt | while read -r sum kind path url; do \
      mkdir -p "$(dirname "$path")"; \
      curl -fsSL --retry 5 --retry-delay 3 -o /tmp/docs.dl "$url"; \
      case "$kind" in \
        file) mv /tmp/docs.dl "$path" ;; \
        md) awk 'NR == 1 && $0 == "---" { fm = 1; next } fm && $0 == "---" { fm = 0; next } !fm' \
              /tmp/docs.dl > "$path"; rm -f /tmp/docs.dl ;; \
        *) echo "unknown kind: $kind" >&2; exit 1 ;; \
      esac; \
      printf '%s  %s\n' "$sum" "$path" >> /tmp/docs.sha256; \
    done; \
    test "$(wc -l < /tmp/docs.sha256)" -eq "$(grep -c "$re" SOURCES.txt)"; \
    sha256sum -c --strict --quiet /tmp/docs.sha256; \
    chmod -R a+rX /opt/docs

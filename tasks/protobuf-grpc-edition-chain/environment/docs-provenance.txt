How environment/docs/ was produced: the build stage below fetches each
document from its official source, checks it against the pinned sha256,
and makes the text conversions. The resulting files are stored in docs/
and copied into the image directly; re-run these steps to refresh a document.

# Reference documents for an agent that works offline, fetched at build time from
# their publishers and pinned by sha256: protobuf.dev's editions pages (at the docs
# commit current when protoc 32.1 was released), gRPC's HTTP/2 protocol and status
# codes (grpc/grpc v1.84.0), and the HTTP/2 and HPACK RFCs. Documents only: no
# protoc, no protobuf or gRPC sources, no conformance suite. They stay out of /app,
# which the verifier collects and scans.
FROM ubuntu:24.04 AS docs
ARG DEBIAN_FRONTEND=noninteractive
RUN apt-get update \
    && apt-get install -y --no-install-recommends ca-certificates curl \
    && rm -rf /var/lib/apt/lists/*
RUN set -eux; \
    fetch() { \
      mkdir -p "/opt/docs/$1"; \
      curl -fsSLo "/opt/docs/$1/$2" "$3"; \
      echo "$4  $1/$2" >> /opt/docs/SHA256SUMS; \
    }; \
    fetch protobuf edition-2023-spec.md https://raw.githubusercontent.com/protocolbuffers/protocolbuffers.github.io/6020dc5c00519e9dbe93e9005ee8b697f8482236/content/reference/protobuf/edition-2023-spec.md 567b49c0a7023881ac65409bf4c16e4e566ec9b7daddb0197e73491e946a5f83; \
    fetch protobuf edition-2024-spec.md https://raw.githubusercontent.com/protocolbuffers/protocolbuffers.github.io/6020dc5c00519e9dbe93e9005ee8b697f8482236/content/reference/protobuf/edition-2024-spec.md bfaf9d3b721d2581ae4dec4868b3e13107cfe79619e08385c23bff906c5bc8a3; \
    fetch protobuf overview.md https://raw.githubusercontent.com/protocolbuffers/protocolbuffers.github.io/6020dc5c00519e9dbe93e9005ee8b697f8482236/content/editions/overview.md 4019f4b6439ac3a855d24450b1bcb7fdadf07c2f380c306e97e119b4b6f0527c; \
    fetch protobuf features.md https://raw.githubusercontent.com/protocolbuffers/protocolbuffers.github.io/6020dc5c00519e9dbe93e9005ee8b697f8482236/content/editions/features.md 38910aaed0284cb89d93ea74ebd5e7e1a8090f2e9ae30ed84a2280ba17279bcc; \
    fetch protobuf field_presence.md https://raw.githubusercontent.com/protocolbuffers/protocolbuffers.github.io/6020dc5c00519e9dbe93e9005ee8b697f8482236/content/programming-guides/field_presence.md 10576686a0bbb3d5f89936ad07867cc470a26929bed5263515d2515f6ce0e7eb; \
    fetch grpc PROTOCOL-HTTP2.md https://raw.githubusercontent.com/grpc/grpc/3252a89f10d8e92997862167ca7d095ecda85973/doc/PROTOCOL-HTTP2.md b51ea1b8cee67ad3477e73152bb60ac3ebf6e164962b46ba79434e980739256f; \
    fetch grpc statuscodes.md https://raw.githubusercontent.com/grpc/grpc/3252a89f10d8e92997862167ca7d095ecda85973/doc/statuscodes.md 98d827f87849c1ab80b3f5f94e0e8f95278c0ff36a51493323f9ca2382bfaa73; \
    fetch rfc rfc9113.txt https://www.rfc-editor.org/rfc/rfc9113.txt a00ef91b64e111a282e77ec66980f5242e77c0bb5e33e0927e3b6757080506de; \
    fetch rfc rfc7541.txt https://www.rfc-editor.org/rfc/rfc7541.txt 2239d7f8fb839b69ae2e928e685559b11376888269f131512197a0e3bacf7f7a; \
    (cd /opt/docs && sha256sum -c SHA256SUMS); \
    chmod -R a+rX /opt/docs

# The class-file major version this rung targets.
#
# Staged rather than written into the verifier, for one reason: C-1 and the
# step's instruction have to be talking about the same number, and on a
# thirteen-rung ladder the number changes at six of the rungs. A literal in
# test.sh is a literal that gets copied forward and not edited -- which is the
# failure that reads as a pass.
#
# 55 is Java 11. The ladder: 50 = Java 6 (steps 01-02), 51 = Java 7 (03),
# 52 = 8 (04-05), 54 = 10 (06), 55 = 11 (07), 61 = 17, 65 = 21, 69 = 25.
#
# THIS RUNG MOVES IT, 54 -> 55, and that makes the class-file level this step's
# must-flip: C-1 owns it. B-4 -- which at step 06 asked for "one major, and it
# is not below 52" -- is restated here as "one major, and it is not below 54",
# so that it holds the rung below rather than this one and family B still reads
# 1.0 on the tree this step is handed.
#
# 55 and not 61: Java 11 is an LTS, `java.net.http` is standard at 11 and
# incubating at 10, and the jakarta namespace move is an 11-era change. 17 is
# the next rung's business.
#
# The scorer validates this on read and refuses to grade if it is missing or
# implausible, so that a staging mistake cannot become a zero that looks like a
# measurement about the tree.
major=55
